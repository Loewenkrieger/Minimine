cmake_minimum_required(VERSION 3.15)
project(MinimineBench VERSION 1.2.0 LANGUAGES C ASM)
if(NOT "${FXSDK_PLATFORM_LONG}" STREQUAL "fxCP")
  message(FATAL_ERROR "Use the ClassPad fxSDK toolchain")
endif()
include(GenerateHH3)
find_package(Gint 2.11 REQUIRED)
if(NOT IS_DIRECTORY "${AZUR_SOURCE}/azur")
  message(FATAL_ERROR "Pinned Azur source is required")
endif()
set(AZUR_VERSION "0.3")
set(Azur_VERSION "0.3")
set(Azur_VERSION_MAJOR 0)
set(Azur_VERSION_MINOR 3)
set(AZUR_PLATFORM_GENERIC 1)
set(AZUR_TOOLKIT_GINT 1)
set(AZUR_GRAPHICS_GINT_CG 1)
set(AZUR_ARCH_SH4ALDSP 1)
set(AZUR_TERMINAL_NONE 1)
configure_file("${AZUR_SOURCE}/azur/include/azur/config.h.in" generated/azur/config.h)
add_library(bench_azur STATIC "${AZUR_SOURCE}/azur/src/gint/render.c" "${AZUR_SOURCE}/azur/src/gint/r61523.s")
target_include_directories(bench_azur PUBLIC "${AZUR_SOURCE}/azur/include" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(bench_azur PRIVATE -Os -g -ffunction-sections -fdata-sections)
target_link_libraries(bench_azur PUBLIC Gint::Gint)
add_executable(minimine_bench main.c scenes.c graphics.c log.c
  ../src/game.c ../src/items.c ../src/inventory.c ../src/inventory_ui.c
  ../src/adventure.c ../src/boss.c ../src/survival.c ../src/validate.c
  ../src/storage.c ../src/perf.c ../src/render.c)
target_include_directories(minimine_bench PRIVATE ../src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(minimine_bench PRIVATE MINIMINE_BENCH_LIGHT)
target_compile_options(minimine_bench PRIVATE -std=c11 -Wall -Wextra -Wshadow -Os -g -ffunction-sections -fdata-sections)
target_link_options(minimine_bench PRIVATE -Wl,--gc-sections "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/MinimineBench.map")
target_link_libraries(minimine_bench bench_azur Gint::Gint)
generate_hh3(TARGET minimine_bench OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/MinimineBench.hh3")
configure_file(build_id.h.in generated/build_id.h @ONLY)
